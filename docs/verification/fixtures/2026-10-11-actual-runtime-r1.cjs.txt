function inventory(base){
 const result=[];let bytes=0;
 function walk(rel=''){
  for(const name of fs.readdirSync(path.join(base,rel)).sort()){
   const item=rel?rel+'/'+name:name,file=path.join(base,item),stat=fs.lstatSync(file);
   assert.ok(!stat.isSymbolicLink()&&(stat.isFile()||stat.isDirectory()));
   assert.ok(result.length<2048&&item.split('/').length<=12);
   if(stat.isDirectory()){result.push({path:item,kind:'directory'});walk(item);}
   else{bytes+=stat.size;assert.ok(stat.size<=16*1024*1024&&bytes<=128*1024*1024);result.push({path:item,kind:'file',bytes:stat.size,sha256:sha(file)});}
  }
 }walk();return result;
}
